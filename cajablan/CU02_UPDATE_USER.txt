async update(id: number, dto: UpdateUserDto) {
    (1) await this.ensureExists(id);

    (2) const data: Record<string, unknown> = {};

    (3) if (dto.name) data.nombre = dto.name;

    (4) if (dto.lastName !== undefined) data.apellido = dto.lastName;

    (5) if (dto.ci !== undefined) data.ci = dto.ci;

    (6) if (dto.email) data.correo = dto.email.toLowerCase();

    (7) if (dto.rol) data.rol = dto.rol;

    (8) if (dto.password) data.password = await bcrypt.hash(dto.password, 10);

    (9) return this.prisma.usuario.update({ where: { id }, data, select: usuarioSelect });
}
