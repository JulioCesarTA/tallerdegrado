CU16: Configurar Punto de Acceso - Actualizar Punto de Acceso

async updateAccessPoint(id: number, dto: UpdatePuntoAccesoDto) {
    (1) const existing = await this.prisma.puntoAcceso.findUnique({ where: { id } });

    (2) if (!existing) throw new NotFoundException('Punto de acceso no encontrado');

    (3) return this.prisma.puntoAcceso.update({
        where: { id },
        data: dto,
        include: {
            camaraIngreso: true,
            camaraSalida: true,
            usuario: { select: { id: true, nombre: true, correo: true } },
        },
    });
}
