CU01: Obtener Perfil del Usuario (Me)

async me(userId: number) {
    (1) const usuario = await this.prisma.usuario.findUnique({
        where: { id: userId },
        select: { id: true, nombre: true, correo: true, rol: true },
    });

    (2) if (!usuario) throw new UnauthorizedException('Usuario no encontrado');

    (3) return { ...usuario, permissions: [] };
}
