async createAccessPoint(dto: CreatePuntoAccesoDto) {
    try {
        (1) return await this.prisma.puntoAcceso.create({
            data: {
                nombre: dto.nombre,
                ubicacion: dto.ubicacion,
                descripcion: dto.descripcion,
                estado: dto.estado,
                camaraIngresoId: dto.camaraIngresoId,
                camaraSalidaId: dto.camaraSalidaId,
                usuarioId: dto.usuarioId,
            },
            include: {
                camaraIngreso: true,
                camaraSalida: true,
                usuario: { select: { id: true, nombre: true, correo: true } },
            },
        });
    } catch (e: unknown) {
        (2) const code = (e as { code?: string })?.code;

        (3) if (code === 'P2002') throw new ConflictException('Una de las camaras ya esta asignada a otro punto de acceso');

        (4) if (code === 'P2003') throw new NotFoundException('Camara o usuario no encontrado');

        (5) throw e;
    }
}
