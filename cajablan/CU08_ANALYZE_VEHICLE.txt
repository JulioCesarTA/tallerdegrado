async analyzeVehicle(file: Express.Multer.File): Promise<VehicleAnalysis> {
    (1) if (!this.claudeApiKey) {
        throw new InternalServerErrorException('Claude API key no configurada');
    }

    (2) const prompt = `Analiza la imagen del vehiculo y responde SOLO con JSON valido...`;

    (3) const mediaType = (file.mimetype as 'image/jpeg' | 'image/png' | 'image/gif' | 'image/webp') || 'image/jpeg';

    (4) const response = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: { 'x-api-key': this.claudeApiKey, ... },
        body: JSON.stringify({ model: this.claudeModel, max_tokens: 512, messages: [...] }),
    });

    (5) if (!response.ok) {
        const errBody = await response.json().catch(() => ({}));
        if (response.status === 429) {
            throw new InternalServerErrorException('Limite de Claude alcanzado...');
        }
        throw new InternalServerErrorException(errBody?.error?.message ?? 'Error al llamar a Claude');
    }

    (6) const data = await response.json();
    const rawText = data.content?.find((c) => c.type === 'text')?.text ?? '';

    (7) const jsonMatch = rawText.match(/\{[\s\S]*\}/);

    (8) if (!jsonMatch) throw new InternalServerErrorException('Respuesta invalida de Claude');

    (9) const parsed = JSON.parse(jsonMatch[0]);

    (10) return {
        tipoVehiculo:   parsed.tipoVehiculo   || 'desconocido',
        marca:          parsed.marca          || 'desconocido',
        modelo:         parsed.modelo         || 'desconocido',
        color:          parsed.color          || 'desconocido',
        caracteristicas: parsed.caracteristicas || 'Sin observaciones.',
    };
}
